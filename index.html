// Add question editor let questionCount = 0; function addQuestion() { questionCount++; const id = questionCount; const box = document.createElement("div"); box.className = "panel"; box.innerHTML = `

प्रश्न ${id}

`; box.querySelector(".removeQuestion").addEventListener("click", () => { box.remove(); }); $("questionEditors").appendChild(box); } $("addQuestionBtn")?.addEventListener("click", addQuestion); // Publish test $("testForm")?.addEventListener("submit", async event => { event.preventDefault(); if (!window.isAdmin) return; const editors = [...document.querySelectorAll("#questionEditors .panel")]; if (!editors.length) { showMessage("testMessage", "कम से कम एक प्रश्न जोड़ें।"); return; } try { const questions = editors.map(box => { const options = [...box.querySelectorAll(".testOption")] .map(input => input.value.trim()); const question = box.querySelector(".questionText").value.trim(); const correctIndex = Number(box.querySelector(".correctAnswer").value); if (!question || options.some(option => !option)) { throw new Error("सभी प्रश्न और चारों विकल्प भरें।"); } return { question, options, correctIndex }; }); const title = $("testTitle").value.trim(); const durationMinutes = Number($("testDuration").value); if (!title || durationMinutes < 1 || durationMinutes > 300) { throw new Error("टेस्ट का नाम और सही समय भरें।"); } await addDoc(collection(db, "tests"), { title, durationMinutes, questions, published: true, createdAt: serverTimestamp() }); $("testForm").reset(); $("questionEditors").innerHTML = ""; questionCount = 0; showMessage("testMessage", "टेस्ट सफलतापूर्वक Publish हो गया!"); loadAdminTests(); } catch (error) { showMessage("testMessage", error.message); } }); // Admin test list window.loadAdminTests = async function() { if (!window.isAdmin) return; const container = $("adminTestList"); if (!container) return; container.textContent = "Loading..."; try { const snapshot = await getDocs(collection(db, "tests")); if (snapshot.empty) { container.textContent = "अभी कोई टेस्ट नहीं है।"; return; } container.innerHTML = snapshot.docs.map(item => `
${safeText(item.data().title)}

${item.data().durationMinutes} मिनट

`).join(""); container.querySelectorAll("[data-delete-test]").forEach(button => { button.addEventListener("click", async () => { if (!confirm("क्या यह टेस्ट हटाना चाहते हैं?")) return; try { await deleteDoc(doc(db, "tests", button.dataset.deleteTest)); loadAdminTests(); } catch (error) { alert(error.message); } }); }); } catch (error) { container.textContent = error.message; } }; // Student test list window.loadStudentTests = async function() { const containers = [ $("studentTestList"), $("categoryTestList") ].filter(Boolean); if (!containers.length) return; containers.forEach(container => { container.textContent = "Loading tests..."; }); try { const snapshot = await getDocs(collection(db, "tests")); const tests = snapshot.docs .filter(item => item.data().published === true); const html = tests.length ? tests.map(item => `

${safeText(item.data().title)}

समय: ${item.data().durationMinutes} मिनट

`).join("") : "अभी कोई टेस्ट उपलब्ध नहीं है।"; containers.forEach(container => { container.innerHTML = html; container.querySelectorAll("[data-start-test]").forEach(button => { button.addEventListener("click", () => { startTest(button.dataset.startTest); }); }); }); } catch (error) { containers.forEach(container => { container.textContent = error.message; }); } }; // Start test async function startTest(testId) { try { if (!window.currentUser) { alert("पहले Student Login करें।"); showPage("login"); return; } const snapshot = await getDoc(doc(db, "tests", testId)); if (!snapshot.exists() || snapshot.data().published !== true) { alert("यह टेस्ट उपलब्ध नहीं है।"); return; } const test = snapshot.data(); const questions = test.questions || []; if (!questions.length) { alert("इस टेस्ट में प्रश्न नहीं हैं।"); return; } const container = $("testArea"); if (!container) { alert("Test Area नहीं मिला।"); return; } showPage("test"); container.innerHTML = `

${safeText(test.title)}

`; const questionContainer = $("attemptQuestions"); questionContainer.innerHTML = questions.map((question, index) => `

Q${index + 1}. ${safeText(question.question)}

${question.options.map((option, optionIndex) => ` `).join("")}
`).join(""); let remaining = Number(test.durationMinutes) * 60; let submitted = false; const timer = setInterval(() => { remaining--; const minutes = Math.floor(Math.max(remaining, 0) / 60); const seconds = Math.max(remaining, 0) % 60; $("testTimer").textContent = `समय शेष: ${minutes}:${String(seconds).padStart(2, "0")}`; if (remaining <= 0) { clearInterval(timer); submitTest(); } }, 1000); async function submitTest() { if (submitted) return; submitted = true; clearInterval(timer); const answers = questions.map((question, index) => { const selected = document.querySelector( `input[name="question${index}"]:checked` ); return selected ? Number(selected.value) : -1; }); let score = 0; questions.forEach((question, index) => { if (answers[index] === Number(question.correctIndex)) { score++; } }); $("attemptForm").querySelectorAll("input,button").forEach(element => { element.disabled = true; }); $("testResult").innerHTML = `

आपका रिजल्ट

${score} / ${questions.length}

सही उत्तर: ${score}

गलत या छोड़े गए प्रश्न: ${questions.length - score}

`; } $("attemptForm").addEventListener("submit", event => { event.preventDefault(); submitTest(); }); $("testTimer").textContent = `समय शेष: ${Math.floor(remaining / 60)}:${String(remaining % 60).padStart(2, "0")}`; } catch (error) { alert(error.message); } } window.startTest = startTest;
